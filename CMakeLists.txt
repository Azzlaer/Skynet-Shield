cmake_minimum_required(VERSION 3.21)
project(SkynetShield VERSION 0.10.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(SkynetShield SHARED Client/Shield.cpp)
target_include_directories(SkynetShield PUBLIC Common)
target_compile_definitions(SkynetShield PRIVATE SHIELD_BUILD)
if(WIN32)
  target_sources(SkynetShield PRIVATE Client/MemoryGuard.cpp Client/ProcessMonitor.cpp Client/SecureCommunication.cpp)
  target_link_libraries(SkynetShield PRIVATE bcrypt winhttp)
  add_library(ShieldBridge SHARED Bridge/ShieldBridge.cpp)
  target_include_directories(ShieldBridge PUBLIC Bridge)
  target_compile_definitions(ShieldBridge PRIVATE BRIDGE_BUILD)
  target_link_libraries(ShieldBridge PRIVATE winhttp)
  if(MSVC)
    target_compile_options(ShieldBridge PRIVATE /W4 /permissive- /sdl)
  endif()
endif()
if(MSVC)
  target_compile_options(SkynetShield PRIVATE /W4 /permissive- /sdl)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(ShieldSmoke Tests/Smoke.cpp)
  target_link_libraries(ShieldSmoke PRIVATE SkynetShield)
  add_test(NAME ShieldSmoke COMMAND ShieldSmoke)
  if(WIN32)
    add_executable(ShieldMemoryTest Tests/MemoryGuard.cpp)
    target_link_libraries(ShieldMemoryTest PRIVATE SkynetShield)
    add_test(NAME ShieldMemoryTest COMMAND ShieldMemoryTest)
    add_executable(ShieldProcessTest Tests/ProcessMonitor.cpp)
    target_link_libraries(ShieldProcessTest PRIVATE SkynetShield)
    add_test(NAME ShieldProcessTest COMMAND ShieldProcessTest)
    add_executable(ShieldCommunicationTest Tests/SecureCommunication.cpp)
    target_link_libraries(ShieldCommunicationTest PRIVATE SkynetShield)
    add_test(NAME ShieldCommunicationTest COMMAND ShieldCommunicationTest)
    add_executable(ShieldSessionWorkerTest Tests/SessionWorker.cpp Integration/Client/SessionWorker.cpp)
    target_include_directories(ShieldSessionWorkerTest PRIVATE Integration/Client)
    target_link_libraries(ShieldSessionWorkerTest PRIVATE SkynetShield)
    add_test(NAME ShieldSessionWorkerTest COMMAND ShieldSessionWorkerTest)
    add_executable(ShieldBridgeTest Tests/ShieldBridge.cpp)
    target_link_libraries(ShieldBridgeTest PRIVATE ShieldBridge)
    add_test(NAME ShieldBridgeTest COMMAND ShieldBridgeTest)
  endif()
endif()
