cmake_minimum_required(VERSION 3.21)
project(SkynetShield VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(SkynetShield SHARED Client/Shield.cpp)
target_include_directories(SkynetShield PUBLIC Common)
target_compile_definitions(SkynetShield PRIVATE SHIELD_BUILD)
if(WIN32)
  target_sources(SkynetShield PRIVATE Client/MemoryGuard.cpp)
  target_link_libraries(SkynetShield PRIVATE bcrypt)
endif()
if(MSVC)
  target_compile_options(SkynetShield PRIVATE /W4 /permissive- /sdl)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(ShieldSmoke Tests/Smoke.cpp)
  target_link_libraries(ShieldSmoke PRIVATE SkynetShield)
  add_test(NAME ShieldSmoke COMMAND ShieldSmoke)
  if(WIN32)
    add_executable(ShieldMemoryTest Tests/MemoryGuard.cpp)
    target_link_libraries(ShieldMemoryTest PRIVATE SkynetShield)
    add_test(NAME ShieldMemoryTest COMMAND ShieldMemoryTest)
  endif()
endif()
